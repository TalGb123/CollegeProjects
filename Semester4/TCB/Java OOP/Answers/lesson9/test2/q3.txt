public static void count(Object[] arr) {
    int countApple = 0;
    int countFruitNotApple = 0;
    int countNotFruit = 0;

    for (int i = 0; i < arr.length; i++) {
        if (arr[i] instanceof Apple) {
            countApple++;
        } 
        else if (arr[i] instanceof Fruit) {
            countFruitNotApple++;
        } 
        else {
            countNotFruit++;
        }
    }
    System.out.println("Apple: " + countApple);
    System.out.println("Fruit (not Apple): " + countFruitNotApple);
    System.out.println("Not Fruit: " + countNotFruit);
}
