public static Triangle maxArea(Shape[] arr) {
    Triangle maxT = null;
    double maxAreaSize = -1;

    for (Shape s : arr) {
        if (s instanceof Triangle) {
            Triangle temp = (Triangle) s;
            if (temp.area() > maxAreaSize) {
                maxAreaSize = temp.area();
                maxT = temp;
            }
        }
    }
    return maxT;
}
