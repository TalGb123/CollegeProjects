public static int maxDiff(ArrayList<Integer> list) {
    if (list == null || list.isEmpty()) return 0;
    int min = list.get(0);
    int max = list.get(0);
    for (int i = 0; i < list.size(); i++) {
        int val = list.get(i);
        if (val < min) min = val;
        if (val > max) max = val;
    }
    return max - min;
}

public static boolean existDiff(ArrayList<Integer> list, int num) {
    if (list == null) return false;
    for (int i = 0; i < list.size(); i++) {
        for (int j = 0; j < list.size(); j++) {
            if (i == j) continue;
            int diff = Math.abs(list.get(i) - list.get(j));
            if (diff == num) {
                return true;
            }
        }
    }
    return false;
}
