public double price() {
      return super.price() + (this.terraceArea * COST_TERRACE);

Penthouse:
public double price() {
      double p = super.price();
      if (!this.seaView) {
            p = p * 0.95; 
      }
      return p;
}

public static void printAffordable(Appartment[] building, double budget) {
      for (int i = 0; i < building.length; i++) {
            if (building[i].getOwner().equals("FREE") && building[i].price() <= budget) {
                  System.out.println(building[i].toString() + " Price: " + building[i].price());
            }
      }
